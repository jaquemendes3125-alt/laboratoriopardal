# --- LABORATÓRIO VIRTUAL: Módulo de Busca Molecular --- # Execute no Google Colab — processamento na nuvem import requests def pesquisar_molecula(nome_composto): print(f"🔬 Pesquisando: {nome_composto}...\n") url = f"https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name/{nome_composto}/property/MolecularFormula,MolecularWeight,IUPACName/JSON" try: resp = requests.get(url) dados = resp.json() prop = dados['PropertyTable']['Properties'][0] print("✅ RESULTADO:") print(f"Nome IUPAC: {prop.get('IUPACName','N/A')}") print(f"Fórmula: {prop.get('MolecularFormula','N/A')}") print(f"Peso: {prop.get('MolecularWeight','N/A')} g/mol") except Exception as e: print("Erro:", e) # Troque o nome abaixo pesquisar_molecula("cafeína")
